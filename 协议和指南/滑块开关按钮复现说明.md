# iOS 风格滑块开关按钮（ToggleSwitch）复现说明

> 写给需要复用这个按钮的工程师：本文只交付**按钮本身**——一个自绘的滑块开关控件
> （外加可选的文字标签），**不包含**它在本项目里连接的"Markdown 阅览/源码模式切换"
> 业务逻辑。拿到代码即可独立运行、独立接线。

## 1. 它长什么样、是什么

效果图：一行文字标签（如"阅览模式"）+ 右侧一个 iOS 风格滑块开关：

```
阅览模式  [ ○── ]      ← 关：滑块在左，轨道蓝色
源码模式  [ ──○ ]      ← 开：滑块在右，轨道蛋黄色，切换时有 150ms 滑动动画
```

本质是一个 **PySide6（Qt for Python）自绘控件**，继承 `QAbstractButton`，
对外接口与 `QCheckBox` 完全一致（`isChecked()` / `setChecked()` / `toggled` 信号），
所以任何原来用复选框的地方都能直接替换。

- 轨道尺寸：40 × 22 像素，圆角矩形，两端圆形滑块。
- 两态对色：关=蓝色 `#0765d4`，开=蛋黄色 `#f5b301`，滑块白色。
  （本项目用"对色"表达"两种模式的差异"，而不是传统"彩色=开/灰色=关"；
  颜色是构造参数，你可以随便换。）
- 点击切换时滑块有 150ms 的 `InOutCubic` 缓动滑动动画。

## 3. 完整代码（可直接运行）

保存为 `toggle_switch_demo.py`，`python toggle_switch_demo.py` 即可看到一个
带"阅览模式"标签的滑块开关窗口。这段代码是从本项目
`gui/panels/viewer/toggle_switch.py` 原样抽出的控件本体 + 一个最小演示窗口：

```python
"""滑块开关控件（PySide6 自绘）：iOS 风格两态对色 + 滑动动画。

用法与 QCheckBox 一致（isChecked()/setChecked()/toggled），便于替换复选框。
关键坑：动画目标属性必须注册为 Qt Property（Python 内置 @property 无效——
运行时警告 "animate a non-existing property offset"，动画不生效状态直接跳变）。
"""

import sys

from PySide6.QtCore import (
    Property,
    QEasingCurve,
    QPropertyAnimation,
    QRectF,
    QSize,
    Qt,
)
from PySide6.QtGui import QColor, QPainter
from PySide6.QtWidgets import (
    QAbstractButton,
    QApplication,
    QHBoxLayout,
    QLabel,
    QWidget,
)

_TRACK_W = 40      # 轨道宽（像素）
_TRACK_H = 22      # 轨道高（像素）
_KNOB_MARGIN = 3   # 滑块内边距
_ANIM_MS = 150     # 滑动动画时长（毫秒）


class ToggleSwitch(QAbstractButton):
    """滑块开关：checkable，对外接口与 QCheckBox 一致"""

    def __init__(self, parent=None,
                 off_color="#0765d4",   # 关（左）轨道色
                 on_color="#f5b301",    # 开（右）轨道色
                 knob_color="#ffffff"):  # 滑块色
        super().__init__(parent)
        self.setCheckable(True)
        self.setCursor(Qt.CursorShape.PointingHandCursor)
        self._off_color = QColor(off_color)
        self._on_color = QColor(on_color)
        self._knob_color = QColor(knob_color)

        self._offset = 0.0  # 滑块位置比例：0.0（关，左）~ 1.0（开，右）
        self._anim = QPropertyAnimation(self, b"offset", self)
        self._anim.setDuration(_ANIM_MS)
        self._anim.setEasingCurve(QEasingCurve.Type.InOutCubic)

        self.toggled.connect(self._start_anim)

    # --- 动画属性：必须是 Qt Property，Python 内置 property 无效 ---
    def get_offset(self) -> float:
        return self._offset

    def set_offset(self, value: float):
        self._offset = value
        self.update()

    offset = Property(float, get_offset, set_offset)

    def _start_anim(self, checked: bool):
        # 动画对象长期复用（KeepWhenStopped），切换时仅重设起止值
        self._anim.stop()
        self._anim.setStartValue(self._offset)
        self._anim.setEndValue(1.0 if checked else 0.0)
        self._anim.start()

    def sizeHint(self) -> QSize:
        return QSize(_TRACK_W, _TRACK_H)

    def paintEvent(self, event):
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)

        track = QRectF(0.5, 0.5, _TRACK_W - 1, _TRACK_H - 1)
        radius = track.height() / 2

        # 轨道：两态对色
        painter.setPen(Qt.PenStyle.NoPen)
        painter.setBrush(self._on_color if self.isChecked() else self._off_color)
        painter.drawRoundedRect(track, radius, radius)

        # 滑块：圆形，位置由动画属性 _offset 驱动
        knob_d = track.height() - 2 * _KNOB_MARGIN
        travel = track.width() - knob_d - 2 * _KNOB_MARGIN
        knob_x = track.left() + _KNOB_MARGIN + travel * self._offset
        knob = QRectF(knob_x, track.top() + _KNOB_MARGIN, knob_d, knob_d)
        painter.setBrush(self._knob_color)
        painter.drawEllipse(knob)

        painter.end()


# ---------------- 以下仅为演示窗口，可删 ----------------

if __name__ == "__main__":
    app = QApplication(sys.argv)

    window = QWidget()
    window.setWindowTitle("ToggleSwitch 演示")
    row = QHBoxLayout(window)
    row.setSpacing(4)

    label = QLabel("阅览模式")
    switch = ToggleSwitch(window,
                          off_color="#0765d4",   # 关 = 蓝
                          on_color="#f5b301")    # 开 = 蛋黄色
    switch.setToolTip("切换模式")

    # 演示接线：标签文字跟随开关状态变化（你自己的业务逻辑接这里）
    switch.toggled.connect(
        lambda checked: label.setText("源码模式" if checked else "阅览模式"))

    row.addWidget(label)
    row.addWidget(switch)
    window.show()
    sys.exit(app.exec())
```

## 4. 集成到你自己的项目

控件本体只有 `ToggleSwitch` 一个类，零外部依赖（除 PySide6）。集成三步：

1. 把 `ToggleSwitch` 类拷进你的项目（或作为一个独立模块 `toggle_switch.py`）。
2. 像用 `QCheckBox` 一样实例化并放进任意布局：

   ```python
   from toggle_switch import ToggleSwitch

   switch = ToggleSwitch(parent, off_color="#0765d4", on_color="#f5b301")
   layout.addWidget(switch)
   ```

3. 把你自己的业务逻辑接到 `toggled` 信号上：

   ```python
   switch.toggled.connect(self.on_mode_changed)  # checked: bool
   ```

   想换颜色、尺寸，改构造参数或文件顶部的 `_TRACK_W` / `_TRACK_H` /
   `_KNOB_MARGIN` / `_ANIM_MS` 常量即可。

想要"标签 + 开关"的组合（像效果图那样），用一个 `QHBoxLayout` 横排一个
`QLabel` 和一个 `ToggleSwitch`，间距 4px，与本文演示窗口一致。

## 5. 关键坑（务必读）

**动画属性必须注册为 Qt 的 `Property`，不能用 Python 内置的 `@property`。**

`QPropertyAnimation(self, b"offset", self)` 动画的是 Qt 元对象系统里的属性。
如果 `offset` 用 Python 的 `@property` 定义，运行时会警告：

```
QPropertyAnimation: you're trying to animate a non-existing property offset of your QObject
```

动画直接失效，开关状态瞬间跳变没有滑动效果。正确写法：

```python
offset = Property(float, get_offset, set_offset)   # PySide6.QtCore.Property
```

其余注意点：

- `setCheckable(True)` 不能漏，否则点击不切换状态、`toggled` 不发射。
- 动画对象只创建一个并长期复用，每次切换只重设起止值，避免反复创建对象。
- `paintEvent` 里轨道 `QRectF` 留了 0.5px 偏移（`(0.5, 0.5, W-1, H-1)`），
  这是抗锯齿下边缘对齐的经典画法，别改成整数坐标，否则边缘会糊。
